cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(factorpool_market LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
add_library(factorpool_market_core STATIC market.cpp)
set_target_properties(factorpool_market_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(factorpool_market_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(factorpool_market_core PUBLIC cxx_std_20)
target_compile_options(factorpool_market_core PRIVATE -Wall -Wextra -Wpedantic)
Python3_add_library(_factorpool_market MODULE ../binding/market.cpp)
target_link_libraries(_factorpool_market PRIVATE factorpool_market_core)
target_compile_options(_factorpool_market PRIVATE -Wall -Wextra -Wpedantic)
option(FACTORPOOL_MARKET_BUILD_TESTS "Build isolated market core tests" ON)
if(FACTORPOOL_MARKET_BUILD_TESTS)
  add_executable(factorpool_market_tests ../tests/market_test.cpp)
  target_link_libraries(factorpool_market_tests PRIVATE factorpool_market_core)
  target_compile_options(factorpool_market_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_test(NAME factorpool_market_tests COMMAND factorpool_market_tests)
endif()
